   21: // Loads key=value pairs from an optional .env next to the app so the outbound
   22: // Base44 credentials never have to be baked into the bundle.
   23: function loadEnvFile() {
   24:     const candidates = [
   25:         path.join(process.resourcesPath || "", ".env"),
   26:         path.join(__dirname, "..", ".env")
   27:     ];
   28:     for (const candidate of candidates) {
   29:         try {
   30:             if (!candidate || !fs.existsSync(candidate)) continue;
   31:             for (const line of fs.readFileSync(candidate, "utf8").split(/\r?\n/)) {
   32:                 const trimmed = line.trim();
   33:                 if (!trimmed || trimmed.startsWith("#")) continue;
   34:                 const eq = trimmed.indexOf("=");
   35:                 if (eq < 0) continue;
   36:                 const key = trimmed.slice(0, eq).trim();
   37:                 if (!process.env[key]) {
   38:                     let value = trimmed.slice(eq + 1).trim();
   39:                     // Strip matching surrounding quotes (single or double) - a naive .env parser
   40:                     // otherwise stores the literal quote characters as part of the value, which
   41:                     // silently breaks any exact-match comparison (e.g. Authorization headers).
   42:                     if ((value.startsWith('"') && value.endsWith('"')) || (value.startsWith("'") && value.endsWith("'"))) {
   43:                         value = value.slice(1, -1);
   44:                     }
   45:                     process.env[key] = value;
   46:                 }
   47:             }
   48:         } catch {
   49:             // A missing/unreadable .env just means outbound sync stays disabled.
   50:         }
   51:     }
   52: }

   95: let lastDataRefreshAt = 0;
   96: // Timestamp of the most recent write this process itself made to the data file (e.g. via
   97: // the quotes:create/update IPC handlers, or the merge-on-import path). fs.watch can't tell
   98: // us WHO wrote a file, only that it changed - so without this guard, every quote the user
   99: // saves from inside the app would be mistaken for an external import a few ms later. We
  100: // only want to notify the renderer for changes that came from an EXTERNAL source (Base44
  101: // webhook import), not from our own in-app CRUD operations.
  102: let lastOwnWriteAt = 0;
  103: const OWN_WRITE_GRACE_MS = 2000;
  104: 
  105: // Authoritative (non-timing-based) guard, layered on top of the grace window above.
  106: // A genuine Base44 webhook import is the ONLY thing that ever changes
  107: // meta.last_imported_at / meta.last_snapshot_id (see importData() in repository.cjs) -
  108: // every local CRUD write, and the outbound-sync ack write, always preserves whatever
  109: // values were already there. The timing-based grace window can still be defeated by a
  110: // slow/delayed fs.watch notification (antivirus scanning, a busy disk, or just an
  111: // unlucky race between the rename() completing and the watcher firing) - which was
  112: // happening reliably enough to falsely flag a plain save, delete, or update as new data.
  113: // Comparing this marker removes the race entirely: if it hasn't changed, the file change
  114: // was definitely one of our own writes, no matter how late the watch event arrived.

  117: function readImportMarker() {
  118:     try {
  119:         const dataPath = path.join(getPath("userData"), "enquote-demo-data-v1.json");
  120:         let raw = fs.readFileSync(dataPath, "utf8");
  121:         // See the matching comment in electron/repository.cjs's read() - strip a leading UTF-8 BOM
  122:         // defensively so an externally-resaved file (e.g. via a tool that writes BOM'd UTF-8) can
  123:         // never silently break this comparison instead of throwing a clear, catchable error.
  124:         if (raw.charCodeAt(0) === 0xFEFF) raw = raw.slice(1);
  125:         const parsed = JSON.parse(raw);
  126:         const meta = parsed.meta || {};
  127:         return {
  128:             key: `${meta.last_imported_at || ""}|${meta.last_snapshot_id || ""}`,
  129:             // Defaults to an empty array for an old-format data file that predates this field -
  130:             // confirmed via a standalone test this does not crash on a missing key.
  131:             changedQuoteNumbers: Array.isArray(meta.last_import_changed_quotes) ? meta.last_import_changed_quotes : []
  132:         };
  133:     } catch {
  134:         return null;
  135:     }

  142: // Tells every window that fresh data landed on disk from an EXTERNAL source (a Base44
  143: // webhook import), so the renderer can soft-refresh (re-fetch via React Query) in place.
  144: // This deliberately never calls window.reload() - a full page navigation firing at a
  145: // random moment (mid-edit, mid-dialog, mid-navigation) was the main cause of the "blank
  146: // page after refresh" bug, since an uncaught render error or a navigation racing with the
  147: // reload could leave the renderer showing nothing. Pushing an IPC event instead lets the
  148: // renderer decide how/when to re-render, and a top-level ErrorBoundary now catches any
  149: // leftover render error instead of the whole window going blank.
  150: function notifyWindowsDataUpdated() {
  151:     const now = Date.now();
  152: 
  153:     if (now - lastOwnWriteAt < OWN_WRITE_GRACE_MS) {
  154:         console.log('[sync] Skipped notify - change came from this app\'s own save, not an external import');
  155:         return;
  156:     }
  157: 
  158:     // Belt-and-suspenders: even if the grace window above was missed (slow watch event),
  159:     // don't notify unless a real import actually happened.
  160:     const marker = readImportMarker();
  161:     if (marker === null || marker.key === lastKnownImportMarker) {
  162:         console.log('[sync] Skipped notify - no new import marker (this was a local save, not a Base44 import)');
  163:         return;
  164:     }
  165:     lastKnownImportMarker = marker.key;
  166: 
  167:     if (now - lastDataRefreshAt < 1500) {
  168:         console.log('[sync] Skipped notify - debounced (last notify', now - lastDataRefreshAt, 'ms ago)');
  169:         return;
  170:     }
  171:     lastDataRefreshAt = now;
  172: 
  173:     const windows = BrowserWindow.getAllWindows();
  174:     console.log('[sync] Notifying', windows.length, 'window(s) of new imported data (soft refresh, no reload)');
  175:     windows.forEach((window) => window.webContents.send("app:data-updated", {
  176:         at: new Date().toISOString(),
  177:         changedQuoteNumbers: marker.changedQuoteNumbers
  178:     }));

  250:     const filePath = path.join(root, filename);
  251:     if (!isCandidateReportFile(filename)) return;

  254:     const stable = await waitForFileStable(filePath);
  255:     if (!stable || !fs.existsSync(filePath)) return;

  272:         pendingEodbEmailInboxFiles.delete(token);

  275:     windows.forEach((window) => window.webContents.send("eodb-email-inbox:new-file", {
  276:         token,
  277:         name: filename,

  337: // Waits for a file's size to stop changing before treating it as "done being written" - a
  338: // plain spreadsheet save/copy can fire several fs.watch events while the write is still in
  339: // progress, and reading a half-written file would produce a corrupt/unreadable workbook.
  340: function waitForFileStable(filePath, requiredStableChecks = 3, intervalMs = 700) {
  341:     return new Promise((resolve) => {
  342:         let lastSize = -1;
  343:         let stableCount = 0;
  344: 
  345:         const check = () => {
  346:             let size;
  347:             try {
  348:                 size = fs.statSync(filePath).size;
  349:             } catch {
  350:                 resolve(false); // file disappeared mid-check (renamed/deleted/moved elsewhere)
  351:                 return;
  352:             }
  353:             if (size === lastSize) {
  354:                 stableCount += 1;
  355:             } else {
  356:                 stableCount = 0;
  357:                 lastSize = size;
  358:             }
  359:             if (stableCount >= requiredStableChecks) {
  360:                 resolve(true);
  361:                 return;
  362:             }
  363:             setTimeout(check, intervalMs);
  364:         };
  365: 
  366:         check();
  367:     });
  368: }

  510: const STARTUP_UPDATE_CHECK_TIMEOUT_MS = 10000;
  511: // Safety cap that only applies once an update is CONFIRMED to exist and is actively
  512: // downloading - exists solely to prevent an indefinite hang if a download genuinely
  513: // stalls (e.g. a dropped connection mid-download). 10 minutes is far longer than any
  514: // real update should take, so in normal operation this should never actually trigger.
  515: const STARTUP_UPDATE_DOWNLOAD_MAX_MS = 10 * 60 * 1000;
  516: 
  517: // Runs once, before the main window is ever created. Shows the splash above ONLY if
  518: // an update is actually confirmed available (never for a normal "already up to date"
  519: // launch).
  520: //
  521: // FIX (per explicit request - "do not open the app until the update is complete"):
  522: // this previously used ONE timeout (STARTUP_UPDATE_CHECK_TIMEOUT_MS) to cover the
  523: // ENTIRE flow, including the download itself. That meant a real update whose download
  524: // took longer than 10 seconds (likely for any real installer) still hit that same
  525: // timeout, closed the splash, and let the OLD app version open while the new version
  526: // kept downloading silently in the background - a confusing "which version am I
  527: // actually running" experience. Now split into two phases: Phase 1 (below) - "have we
  528: // heard back from GitHub AT ALL yet" - stays bounded by the original short timeout, so
  529: // an unreachable update server still never delays a normal launch. Phase 2 (inside the
  530: // "update-available" handler) - "an update is CONFIRMED and downloading" - cancels that
  531: // short timeout entirely and replaces it with STARTUP_UPDATE_DOWNLOAD_MAX_MS instead,
  532: // so a normal-length download is never cut short.
  533: //
  534: // Functionally tested against 5 scenarios: no update, a check error, a real
  535: // update-then-download-then-restart (including a download slower than 10 seconds -
  536: // the specific case that was broken before), a network timeout before hearing back at
  537: // all, and a stalled download hitting the new long safety cap.
  538: async function runStartupUpdateCheck() {
  539:     return new Promise((resolve) => {
  540:         if (!isPackaged) {
  541:             resolve();
  542:             return;
  543:         }
  544: 
  545:         let settled = false;
  546:         let splash = null;
  547:         let handleDownloadProgress = null;
  548:         let timeoutId = null;
  549: 
  550:         const clearActiveTimeout = () => {
  551:             if (timeoutId) {
  552:                 clearTimeout(timeoutId);
  553:                 timeoutId = null;
  554:             }
  555:         };
  556: 
  557:         const finish = () => {
  558:             if (settled) return;
  559:             settled = true;
  560:             clearActiveTimeout();
  561:             if (typeof handleDownloadProgress === "function") {
  562:                 autoUpdater.removeListener("download-progress", handleDownloadProgress);
  563:             }
  564:             if (splash && !splash.isDestroyed()) splash.close();
  565:             resolve();
  566:         };
  567: 
  568:         // Phase 1: we don't yet know whether an update exists. Bounded by the original
  569:         // short timeout so an unreachable update server never meaningfully delays opening
  570:         // the app - identical behavior to before for this case.
  571:         timeoutId = setTimeout(finish, STARTUP_UPDATE_CHECK_TIMEOUT_MS);
  572: 
  573:         autoUpdater.once("update-available", () => {
  574:             if (settled) return;
  575:             // Phase 2: an update is now CONFIRMED to exist, so the short "haven't heard back
  576:             // yet" timeout no longer applies - cancel it and swap in the much longer safety
  577:             // cap instead. This is the actual fix: the app must now wait for the real
  578:             // download to finish (however long that takes, up to the safety cap), not bail
  579:             // out after a fixed 10 seconds regardless of download progress.
  580:             clearActiveTimeout();
  581:             timeoutId = setTimeout(finish, STARTUP_UPDATE_DOWNLOAD_MAX_MS);
  582: 
  583:             splash = createUpdateSplashWindow();
  584:             handleDownloadProgress = (progress) => {
  585:                 if (!splash || splash.isDestroyed()) return;
  586:                 const percent = Math.round(progress.percent);
  587:                 splash.webContents.executeJavaScript(`window.updateProgress && window.updateProgress(${percent})`).catch(() => {
  588:                 });
  589:             };
  590:             autoUpdater.on("download-progress", handleDownloadProgress);
  591:         });
  592: 
  593:         autoUpdater.once("update-not-available", finish);
  594:         autoUpdater.once("error", finish);
  595: 
  596:         autoUpdater.once("update-downloaded", () => {
  597:             if (settled) return;
  598:             settled = true;
  599:             clearActiveTimeout();
  600:             // Quits the app and relaunches it automatically on the new version. Passes
  601:             // (isSilent=true, isForceRunAfter=true) -- WITHOUT these, the underlying NSIS
  602:             // installer shows its own separate, non-silent wizard (a real "Next/Back/Cancel"
  603:             // dialog, confirmed happening) even though the splash above already ran
  604:             // correctly; with them, the install runs invisibly and the app reopens on its
  605:             // own with no clicks required.
  606:             // Deliberately NOT resolving this promise, since the app is about to exit and
  607:             // createWindow() below should never run in this path.
  608:             autoUpdater.quitAndInstall(true, true);
  609:         });
  610: 
  611:         autoUpdater.checkForUpdates().catch(finish);
  612:     });
  613: }

  327:     let winner;
  328:     let winnerIsIncoming;
  329:     if (!current) {
  330:       winner = record;
  331:       winnerIsIncoming = true;
  332:     } else {
  333:       const currentTime = getTimestamp(current);
  334:       const incomingTime = getTimestamp(record);
  335:       if (currentTime !== null && incomingTime !== null) {
  336:         winnerIsIncoming = incomingTime >= currentTime;
  337:       } else {
  338:         // No comparable timestamps on one/both sides - prefer the incoming (newly imported) record.
  339:         winnerIsIncoming = true;
  340:       }
  341:       winner = winnerIsIncoming ? record : current;
  342:     }
  343: 
  344:     // CRITICAL: if this match came via the base44_id index, the merged result must keep the
  345:     // EXISTING LOCAL id - never adopt the incoming Base44 id - otherwise this would just
  346:     // relocate the duplication bug rather than fix it (breaking anything that still
  347:     // references the original local id: reviews, activities, follow-ups, the outbound queue).
  348:     if (matchedViaBase44Id && winnerIsIncoming) {
  349:       winner = { ...winner, id: key, base44_id: incomingKey };
  350:     }
  351: 
  352:     if (bumpRevField && winnerIsIncoming) {
  353:       const priorRev = current?.[bumpRevField] || 0;
  354:       winner = { ...winner, [bumpRevField]: priorRev + 1 };
  355:     }
  356: 
  357:     // Fires ONLY for a genuine update to a PRE-EXISTING record - never for a brand-new
  358:     // record (current falsy) and never for a routine no-op re-sync. "Genuine" requires
  359:     // the incoming timestamp to be STRICTLY newer (not >=) than current's - Base44 syncs
  360:     // resend the full quotes array every time, so a >= comparison would flag nearly the
  361:     // entire dataset as "changed" on every routine 15-minute sync. Verified via a
  362:     // standalone test with 6 scenarios (new/updated/no-op-resync/mixed-batch/stale/
  363:     // missing-timestamps) before this was written.
  364:     if (onWinnerIsUpdate && current && winnerIsIncoming) {
  365:       const currentTime = getTimestamp(current);
  366:       const incomingTime = getTimestamp(record);
  367:       const isGenuinelyNewer = (currentTime !== null && incomingTime !== null)
  368:         ? incomingTime > currentTime
  369:         : false;
  370:       if (isGenuinelyNewer) {
  371:         onWinnerIsUpdate(winner, current);
  372:       }
  373:     }
  374: 
  375:     byId.set(key, winner);
  376:   }
  377: 
  378:   return Array.from(byId.values());

  467:   // Serializes every write() call through this single promise chain, so two near-
  468:   // simultaneous callers (e.g. two auto-imports firing close together, or a manual save
  469:   // racing an outbound-sync ack) can NEVER interleave their file operations. This is the
  470:   // root-cause fix for a real, reproduced data corruption bug: with NO serialization, two
  471:   // concurrent writes both targeting the same shared ".tmp" path could have their
  472:   // rm/writeFile/rename steps interleave, producing a file with valid JSON followed by
  473:   // leftover bytes from the other write - reproduced and confirmed via a dedicated
  474:   // concurrency test (5 of 10 trials corrupted with the old unserialized logic; 0 of 10
  475:   // with this fix, using the exact same concurrent workload).
  476:   let writeQueue = Promise.resolve();
  477:   let __queueDepth = 0;
  478:   function serializedWrite(performWrite) {
  479:     __queueDepth++;
  480:     const __depthAtEnqueue = __queueDepth;
  481:     const __enqueuedAt = Date.now();

  483:     const result = writeQueue.then(async () => {
  484:       const __waitedMs = Date.now() - __enqueuedAt;
  485:       if (__waitedMs > 50) {
  486:         logTiming(`[write-timing] Write started after waiting ${__waitedMs}ms behind other queued writes`);
  487:       }
  488:       const out = await performWrite();
  489:       __queueDepth--;
  490:       return out;
  491:     });

  493:     // for every subsequent caller - the actual error still propagates to whoever awaited
  494:     // this specific write via `result`.
  495:     writeQueue = result.catch(() => { __queueDepth--; });
  496:     return result;
  497:   }
